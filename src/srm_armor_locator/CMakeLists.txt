cmake_minimum_required(VERSION 3.8)
project(srm_armor_locator)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(srm_aim_interfaces REQUIRED)
find_package(srm_armor_refiner REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs calib3d)
add_library(locator_core src/locator.cpp)
target_compile_features(locator_core PUBLIC cxx_std_17)
target_include_directories(locator_core PUBLIC include ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(locator_core srm_armor_refiner)
target_link_libraries(locator_core ${OpenCV_LIBS} yaml-cpp)
target_compile_options(locator_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(armor_locator_node src/armor_locator_node.cpp)
ament_target_dependencies(armor_locator_node rclcpp rcl_interfaces ament_index_cpp sensor_msgs
  std_msgs diagnostic_msgs geometry_msgs cv_bridge srm_aim_interfaces srm_armor_refiner)
target_link_libraries(armor_locator_node locator_core)
target_compile_options(armor_locator_node PRIVATE -Wall -Wextra -Wpedantic)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_locator test/test_locator.cpp)
  target_link_libraries(test_locator locator_core)
  target_compile_definitions(test_locator PRIVATE
    TEMPLATE_PATH="${CMAKE_CURRENT_SOURCE_DIR}/templates/digit_2_template.png")
endif()
install(TARGETS locator_core ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(TARGETS armor_locator_node DESTINATION lib/${PROJECT_NAME})
install(FILES templates/digit_2_template.png
  DESTINATION share/${PROJECT_NAME}/templates)
ament_package()
