cmake_minimum_required(VERSION 3.8)
project(srm_armor_tracker)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(srm_aim_interfaces REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core)
add_library(tracker_core src/tracker.cpp)
target_compile_features(tracker_core PUBLIC cxx_std_17)
target_include_directories(tracker_core PUBLIC include ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(tracker_core srm_aim_interfaces)
target_link_libraries(tracker_core ${OpenCV_LIBS})
target_compile_options(tracker_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(armor_tracker_node src/armor_tracker_node.cpp)
ament_target_dependencies(armor_tracker_node rclcpp rcl_interfaces diagnostic_msgs srm_aim_interfaces)
target_link_libraries(armor_tracker_node tracker_core)
target_compile_options(armor_tracker_node PRIVATE -Wall -Wextra -Wpedantic)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_tracker test/test_tracker.cpp)
  target_link_libraries(test_tracker tracker_core)
endif()
install(TARGETS tracker_core ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(TARGETS armor_tracker_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_package()
