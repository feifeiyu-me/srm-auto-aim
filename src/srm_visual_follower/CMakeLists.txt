cmake_minimum_required(VERSION 3.8)
project(srm_visual_follower)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(srm_aim_interfaces REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(builtin_interfaces REQUIRED)
add_library(follower_core src/follower.cpp)
target_compile_features(follower_core PUBLIC cxx_std_17)
target_include_directories(follower_core PUBLIC include)
ament_target_dependencies(follower_core srm_aim_interfaces geometry_msgs std_msgs builtin_interfaces)
target_compile_options(follower_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(visual_follower_node src/visual_follower_node.cpp)
ament_target_dependencies(visual_follower_node rclcpp rcl_interfaces diagnostic_msgs srm_aim_interfaces)
target_link_libraries(visual_follower_node follower_core)
target_compile_options(visual_follower_node PRIVATE -Wall -Wextra -Wpedantic)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_follower test/test_follower.cpp)
  target_link_libraries(test_follower follower_core)
endif()
install(TARGETS follower_core ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(TARGETS visual_follower_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_package()
