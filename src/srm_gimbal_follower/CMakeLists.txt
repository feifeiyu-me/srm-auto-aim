cmake_minimum_required(VERSION 3.8)
project(srm_gimbal_follower)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(std_srvs REQUIRED)
add_library(gimbal_follower_core src/controller.cpp)
target_compile_features(gimbal_follower_core PUBLIC cxx_std_17)
target_include_directories(gimbal_follower_core PUBLIC include)
ament_target_dependencies(gimbal_follower_core diagnostic_msgs geometry_msgs std_msgs builtin_interfaces)
target_compile_options(gimbal_follower_core PRIVATE -Wall -Wextra -Wpedantic)
foreach(node gimbal_follower_node gimbal_simulator_node manual_step_node)
  add_executable(${node} src/${node}.cpp)
  ament_target_dependencies(${node} rclcpp rcl_interfaces diagnostic_msgs geometry_msgs std_msgs std_srvs)
  target_link_libraries(${node} gimbal_follower_core)
  target_compile_options(${node} PRIVATE -Wall -Wextra -Wpedantic)
  install(TARGETS ${node} DESTINATION lib/${PROJECT_NAME})
endforeach()
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_controller test/test_controller.cpp)
  target_link_libraries(test_controller gimbal_follower_core)
  ament_add_gtest(test_manual_step test/test_manual_step.cpp)
  target_link_libraries(test_manual_step gimbal_follower_core)
endif()
install(TARGETS gimbal_follower_core ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME}
  PATTERN "__pycache__" EXCLUDE)
ament_package()
